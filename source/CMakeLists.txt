cmake_minimum_required(VERSION 3.16)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/version.cmake")
project(RealCUGANDesktop VERSION ${RCG_VERSION} LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
option(RCG_GPU "Build the Vulkan worker (CPU worker is built in a separate build directory)" ON)
set(NCNN_VULKAN ${RCG_GPU} CACHE BOOL "" FORCE)
if(RCG_GPU AND NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ncnn/glslang/CMakeLists.txt")
    message(FATAL_ERROR "Missing pinned glslang submodule. See BUILD.md; do not substitute a different ncnn or model.")
endif()
set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NCNN_OPENMP ON CACHE BOOL "" FORCE)
set(NCNN_STATIC_CRT ON CACHE BOOL "" FORCE)
set(NCNN_DISABLE_EXCEPTION OFF CACHE BOOL "" FORCE)
set(NCNN_SSE2 ON CACHE BOOL "" FORCE)
set(NCNN_AVX ON CACHE BOOL "" FORCE)
set(NCNN_AVX2 ON CACHE BOOL "" FORCE)
set(NCNN_AVX512 OFF CACHE BOOL "" FORCE)
set(NCNN_FMA ON CACHE BOOL "" FORCE)
set(NCNN_RUNTIME_CPU ON CACHE BOOL "" FORCE)
set(NCNN_XOP OFF CACHE BOOL "" FORCE)
set(NCNN_F16C OFF CACHE BOOL "" FORCE)
set(NCNN_AVXVNNI OFF CACHE BOOL "" FORCE)
if(MINGW)
    # 将 OpenMP 运行库静态链接，便携包无需额外安装或携带 libgomp DLL。
    execute_process(COMMAND ${CMAKE_CXX_COMPILER} -print-file-name=libgomp.a OUTPUT_VARIABLE STATIC_GOMP OUTPUT_STRIP_TRAILING_WHITESPACE ENCODING UTF-8)
    set(OpenMP_gomp_LIBRARY "${STATIC_GOMP}" CACHE FILEPATH "Static OpenMP runtime" FORCE)
    get_filename_component(RCG_COMPILER_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
    if(EXISTS "${RCG_COMPILER_BIN}/../lib/libmingwthrd.a")
        set(OpenMP_mingwthrd_LIBRARY "${RCG_COMPILER_BIN}/../lib/libmingwthrd.a" CACHE FILEPATH "Windows thread runtime" FORCE)
    endif()
endif()
add_subdirectory(third_party/ncnn)
if(RCG_GPU AND MINGW)
    # 固定的旧 glslang 头使用 uint32_t/uint64_t，但没有自行包含标准整数头。
    # 从构建端补齐声明，保持第三方源码和着色器逻辑原样。
    foreach(glslang_target IN ITEMS MachineIndependent SPIRV glslang)
        if(TARGET ${glslang_target})
            target_compile_options(${glslang_target} PRIVATE -include cstdint)
        endif()
    endforeach()
endif()
if(RCG_GPU)
    set(WORKER realcugan-gpu-worker)
else()
    set(WORKER realcugan-worker)
endif()
add_executable(${WORKER} worker.cpp realcugan.cpp)
target_include_directories(${WORKER} PRIVATE third_party/fmt/include)
target_compile_definitions(${WORKER} PRIVATE FMT_HEADER_ONLY NOMINMAX UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(${WORKER} PRIVATE ncnn ole32)
add_executable(RealCUGAN WIN32 desktop.cpp "${RCG_GENERATED_DIR}/resources.rc")
target_include_directories(RealCUGAN PRIVATE "${RCG_GENERATED_DIR}")
target_compile_definitions(RealCUGAN PRIVATE NOMINMAX UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(RealCUGAN PRIVATE setupapi advapi32 dxgi pdh psapi d3d11 d2d1 dwrite gdiplus comdlg32 shell32 ole32 uuid comctl32)
if(MINGW)
    target_compile_options(RealCUGAN PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Werror)
    target_link_options(RealCUGAN PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(${WORKER} PRIVATE -municode -static -static-libgcc -static-libstdc++)
elseif(MSVC)
    set_property(TARGET RealCUGAN ${WORKER} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(RealCUGAN PRIVATE /utf-8)
    target_compile_options(${WORKER} PRIVATE /utf-8)
endif()
